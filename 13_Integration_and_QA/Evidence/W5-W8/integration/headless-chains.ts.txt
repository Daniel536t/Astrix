let hseq = 0;
const send = (s: GameSession, c: Record<string, unknown>) => s.dispatch({ schema: 3, id: `hl:${hseq++}:${String(c.type)}`, expectedRevision: s.world.revision, ...c } as never);
const tick = (s: GameSession, n = 1) => { for (let i = 0; i < n; i++) ok(send(s, { type: 'tick', ticks: 1 }), 'tick'); };
function swing(s: GameSession, c: Record<string, unknown>) { for (let i = 0; i < 3; i++) { const r = send(s, c); if (r.ok || code(r) !== 'too_fast') return r; tick(s); } throw new Error('rate limit never cleared'); }


class MemoryStorage implements KeyValueStorage {
  readonly map = new Map<string, string>();
  getItem(k: string) { return this.map.get(k) ?? null; }
  setItem(k: string, v: string) { this.map.set(k, v); }
  removeItem(k: string) { this.map.delete(k); }
  key(i: number) { return [...this.map.keys()][i] ?? null; }
  get length() { return this.map.size; }
}

const explore = (p: P) => send(s, { type: 'explore', at: dm(p) });
const ground = (x: number, z: number) => terrainCache(1337, G).h(Math.floor(x), Math.floor(z));
/** save + reload through the browser store: returns {equal, hash} and continues in the reloaded session. */
function reload(label: string) {
  const before = s.stateHash;
  if (!store().save(s).ok) throw new Error('save ' + label);
  const again = store().load();
  const eq = again.session.stateHash === before;
  s = again.session;
  const checks = (R.reloads as unknown[] | undefined) ?? (R.reloads = []);
  (checks as unknown[]).push({ label, before, after: s.stateHash, equal: eq, status: again.status, records: wl().animals.size, dead: wl().dead.size, carcasses: wl().carcasses.size });
  return eq;
}
function walk(a: P, b: P, withPresence = true) {
  const n = Math.max(1, Math.ceil(Math.hypot(b.x - a.x, b.z - a.z) / 16));
  for (let k = 1; k <= n; k++) { const p = { x: a.x + (b.x - a.x) * k / n, z: a.z + (b.z - a.z) * k / n }; if (withPresence) tickP(p); else tick(s); ok(explore(p), 'report'); }
}
// ---- hunting helpers (same maths as tests/w8.hunting.test.ts) ----
const EYE = 1.4;
function aimAt(from: V, target: V, draw: number, kind: 'arrow' | 'blunt_arrow' = 'arrow') {
  const dx = target.x - from.x, dz = target.z - from.z, D = Math.hypot(dx, dz), hx = dx / D, hz = dz / D;
  const yAt = (e: number) => {
    const dir = quantiseDir(hx * Math.cos(e), Math.sin(e), hz * Math.cos(e));
    const f = launch({ x: Math.round(from.x * 100), y: Math.round(from.y * 100), z: Math.round(from.z * 100) }, dir, draw)!;
    const p = flightPath(f, kind);
    for (let i = 3; i < p.length; i += 3) {
      const d0 = Math.hypot(p[i - 3] - f.x, p[i - 1] - f.z), d1 = Math.hypot(p[i] - f.x, p[i + 2] - f.z);
      if (d1 >= D) { const k = (D - d0) / (d1 - d0); return { y: p[i - 2] + (p[i + 1] - p[i - 2]) * k, dir, secs: (i / 3 - 1 + k) * ARROW_DT }; }
    }
    return { y: -1e9, dir, secs: 99 };
  };
  let lo = -0.7, hi = 0.7;
  for (let i = 0; i < 50; i++) { const m = (lo + hi) / 2; if (yAt(m).y < target.y) lo = m; else hi = m; }
  const r = yAt((lo + hi) / 2); return { dir: r.dir, secs: r.secs };
}
const bodyAt = (a: Animal, frac: number): V => { const pose = animalPose(a, frac), sh = hitShape(a, pose, ground(pose.x, pose.z)); return { x: (sh.a.x + sh.b.x) / 2, y: sh.a.y, z: (sh.a.z + sh.b.z) / 2 }; };
const headAt = (a: Animal, frac: number): V => { const pose = animalPose(a, frac); return hitShape(a, pose, ground(pose.x, pose.z)).head; };
function shootAt(sess: GameSession, stand: P, target: V | ((secs: number) => V), o: { draw?: number; t?: number; arrow?: 'arrow' | 'blunt_arrow' } = {}) {
  const draw = o.draw ?? DRAW_FULL_MS, arrow = o.arrow ?? 'arrow', t = o.t ?? 300;
  const from = { x: stand.x, y: ground(stand.x, stand.z) + EYE, z: stand.z };
  let goal = typeof target === 'function' ? target(0) : target;
  let aim = aimAt(from, goal, draw, arrow);
  if (typeof target === 'function') for (let i = 0; i < 4; i++) { goal = target(aim.secs); aim = aimAt(from, goal, draw, arrow); }
  return send(sess, { type: 'shoot', arrow, from: { x: Math.round(from.x * 100), y: Math.round(from.y * 100), z: Math.round(from.z * 100) }, dir: aim.dir, draw, t, at: dm(stand) });
}
function clearLine(p: P, a: Animal): boolean {
  const tc = terrainCache(1337, G), b = bodyAt(a, 0), y0 = ground(p.x, p.z) + EYE, L = Math.hypot(b.x - p.x, b.z - p.z);
  for (let d = 0.3; d < L - 0.8; d += 0.25) { const k = d / L, x = p.x + (b.x - p.x) * k, z = p.z + (b.z - p.z) * k, y = y0 + (b.y - y0) * k; if (tc.h(Math.floor(x), Math.floor(z)) > y - 0.12 || tc.d(Math.floor(x), Math.floor(z)) > 0) return false; }
  return true;
}
function standOff(a: Animal, d: number, toward: P = SPAWN): P {
  const ax = a.x / 10, az = a.z / 10;
  for (let k = 0; k < 32; k++) { const ang = Math.atan2(toward.z - az, toward.x - ax) + (k % 2 ? 1 : -1) * Math.ceil(k / 2) * 0.2; const p = { x: ax + Math.cos(ang) * d, z: az + Math.sin(ang) * d }; if (terrainCache(1337, G).d(Math.floor(p.x), Math.floor(p.z)) === 0 && clearLine(p, a)) return p; }
  throw new Error('no spot');
}
